<?php

$password = "Singham@11";             // Your database password
$dbname = "awsdatabase-35303337a502";    // Your database name



$conn = new mysqli($host, $username, $password, $dbname);

// Check connection
if ($conn->connect_error) {
    die("Connection failed: " . $conn->connect_error);
}

// Check if the form is submitted
if ($_SERVER["REQUEST_METHOD"] == "POST") {
    // Sanitize and validate input
    $email = $conn->real_escape_string($_POST['email']);
    $password = $conn->real_escape_string($_POST['password']);
    $actkey = $conn->real_escape_string($_POST['actkey']);

    // Query to check email, password, and actkey
    $sql = "SELECT * FROM users WHERE email = '$email' AND actkey = '$actkey'";
    $result = $conn->query($sql);

    if ($result->num_rows > 0) {
        // Fetch user data
        $row = $result->fetch_assoc();

        // Verify password
        if (password_verify($password, $row['password'])) {
            // Successful login
            header("Location: Homepage.html");
            exit;
        } else {
            // Incorrect password
            echo "<p style='color:red;'>Invalid email, password, or activation key.</p>";
        }
    } else {
        // Email or activation key not found
        echo "<p style='color:red;'>Invalid email, password, or activation key.</p>";
    }
}

// Close connection
$conn->close();

           <script>
        // Function to fetch the user count from user_count.php
        function getUserCount() {
            // Create a new XMLHttpRequest object
            var xhr = new XMLHttpRequest();
            
            // Configure it: GET-request for the user_count.php file
            xhr.open('GET', 'user_count.php', true);
            
            // Set up a function that is called when the request is complete
            xhr.onload = function () {
                if (xhr.status == 200) {
                    // On successful response, update the page with the user count
                    document.getElementById('user-count').textContent = xhr.responseText;
                }
            };
            
            // Send the request to the server
            xhr.send();
        }

        // Call the function when the page loads
        window.onload = function() {
            getUserCount();
        };
    </script>
